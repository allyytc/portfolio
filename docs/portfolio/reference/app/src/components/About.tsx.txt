import { useEffect, useRef } from 'react'
import gsap from 'gsap'
import { ScrollTrigger } from 'gsap/ScrollTrigger'

gsap.registerPlugin(ScrollTrigger)

const SKILLS = [
  'PYTHON', 'TYPESCRIPT', 'REACT', 'NEXT.JS', 'PYTORCH', 'OPENCV',
  'FIGMA', 'NODE.JS', 'FLUTTER', 'SCIKIT-LEARN', 'MEDIAPIPE', 'TAILWIND',
  'JAVASCRIPT', 'SQL', 'DART', 'C++', 'GIT', 'DRUPAL', 'GODOT',
]

const TICKER_TEXT = SKILLS.join(' · ') + ' · '

export default function About() {
  const sectionRef = useRef<HTMLElement>(null)
  const quoteRef   = useRef<HTMLParagraphElement>(null)
  const photoRef   = useRef<HTMLDivElement>(null)
  const boxRef     = useRef<HTMLDivElement>(null)
  const prefersReduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches

  useEffect(() => {
    if (prefersReduced || !sectionRef.current) return

    const lines = quoteRef.current?.querySelectorAll('.quote-line')
    if (lines) {
      gsap.fromTo(lines,
        { clipPath: 'inset(0 0 100% 0)', opacity: 0 },
        {
          clipPath: 'inset(0 0 0% 0)', opacity: 1,
          stagger: 0.15, duration: 0.7, ease: 'power2.out',
          scrollTrigger: {
            trigger: sectionRef.current,
            start: 'top 65%',
          },
        }
      )
    }

    gsap.fromTo(photoRef.current,
      { opacity: 0 },
      {
        opacity: 1, duration: 0.8, ease: 'power2.out',
        scrollTrigger: { trigger: sectionRef.current, start: 'top 60%' },
        onComplete: () => {
          gsap.fromTo(boxRef.current,
            { scaleX: 0, transformOrigin: 'top left' },
            { scaleX: 1, duration: 0.4, ease: 'power2.out' }
          )
        },
      }
    )
  }, [])

  return (
    <section
      ref={sectionRef}
      id="about"
      style={{
        background: 'var(--bg)',
        overflow: 'hidden',
      }}
    >
      {/* Ticker */}
      <div style={{
        borderTop: '1px solid var(--border-col)',
        borderBottom: '1px solid var(--border-col)',
        padding: '0.8rem 0',
        overflow: 'hidden',
        whiteSpace: 'nowrap',
      }}>
        <div style={{
          display: 'inline-block',
          animation: 'ticker 28s linear infinite',
          fontFamily: 'var(--font-label)',
          fontSize: 'var(--text-label)',
          color: 'var(--text-dim)',
          letterSpacing: '0.15em',
        }}>
          {TICKER_TEXT}{TICKER_TEXT}{TICKER_TEXT}
        </div>
      </div>

      {/* Content */}
      <div style={{
        display: 'grid',
        gridTemplateColumns: '1.2fr 0.8fr',
        gap: 'clamp(2rem, 6vw, 8rem)',
        padding: 'clamp(4rem, 8vw, 8rem) clamp(1.5rem, 6vw, 6rem)',
        alignItems: 'center',
      }} className="about-grid">
        {/* Pull quote */}
        <p
          ref={quoteRef}
          style={{
            fontFamily: 'var(--font-display)',
            fontSize: 'var(--text-section)',
            color: 'var(--text-primary)',
            lineHeight: 1.3,
            margin: 0,
          }}
        >
          <span className="quote-line" style={{ display: 'block' }}>&ldquo;I work end-to-end -</span>
          <span className="quote-line" style={{ display: 'block' }}>from wireframes to</span>
          <span className="quote-line" style={{ display: 'block' }}>working code - and I&rsquo;m</span>
          <span className="quote-line" style={{ display: 'block' }}>as comfortable in Figma</span>
          <span className="quote-line" style={{ display: 'block' }}>as I am in a codebase.&rdquo;</span>
        </p>

        {/* CV photo */}
        <div
          ref={photoRef}
          style={{
            position: 'relative',
            opacity: 0,
          }}
        >
          <img
            src="/headshot.jpg"
            alt="Ally Chen"
            style={{
              width: '100%',
              aspectRatio: '3/4',
              objectFit: 'cover',
              objectPosition: 'top',
              display: 'block',
              filter: 'grayscale(100%) contrast(1.1)',
            }}
          />
          {/* Bounding box overlay */}
          <div
            ref={boxRef}
            style={{
              position: 'absolute',
              inset: '10%',
              border: '1px solid var(--accent)',
              pointerEvents: 'none',
            }}
          >
            <span style={{
              position: 'absolute',
              top: -20, left: 0,
              fontFamily: 'var(--font-label)',
              fontSize: '9px',
              color: 'var(--accent)',
              letterSpacing: '0.1em',
            }}>PERSON: 0.99</span>
          </div>
        </div>
      </div>

      <style>{`
        @keyframes ticker {
          from { transform: translateX(0); }
          to   { transform: translateX(-33.33%); }
        }
        @media (max-width: 768px) {
          .about-grid { grid-template-columns: 1fr !important; }
          .about-grid > :first-child { order: 2; }
          .about-grid > :last-child  { order: 1; }
        }
      `}</style>
    </section>
  )
}
